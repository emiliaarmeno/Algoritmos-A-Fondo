using namespace std;

struct Presta
{
	int idPres;
	int idPrac;
	long long fechaHora;
	int minutos;
};

struct Practica
{
	int idPrac;
	char descr[50];
};

string prestaToString(Presta x)
{
	char sep = 1;
	string sIdPres=to_string(x.idPres);
	string sIdPrac=to_string(x.idPrac);
	string sFechaHora=to_string(x.fechaHora);
	string sMinutos=to_string(x.minutos);
	return sIdPres+sep+sIdPrac+sep+sFechaHora+sep+sMinutos;
}

Presta prestaFromString(string s)
{
	char sep = 1;
	Presta x;
	string t0 = getTokenAt(s,sep,0);
	x.idPres=stoi(t0);
	string t1 = getTokenAt(s,sep,1);
	x.idPrac=stoi(t1);
	string t2 = getTokenAt(s,sep,2);
	x.fechaHora=atoll(t2.c_str());
	string t3 = getTokenAt(s,sep,3);
	x.minutos=stoi(t3);
	return x;
}

string prestaToDebug(Presta x)
{
	stringstream sout;
	sout<< "[";
	sout << x.idPres;
	sout << ",";
	sout << x.idPrac;
	sout << ",";
	sout << x.fechaHora;
	sout << ",";
	sout << x.minutos;
	sout<< "]";
	return sout.str();
}

Presta presta(int idPres,int idPrac,long long fechaHora,int minutos)
{
	Presta a;
	a.idPres = idPres;
	a.idPrac = idPrac;
	a.fechaHora = fechaHora;
	a.minutos = minutos;
	return a;
}

bool prestaEquals(Presta a,Presta b)
{
	if(a.idPres!=b.idPres) return false;
	if(a.idPrac!=b.idPrac) return false;
	if(a.fechaHora!=b.fechaHora) return false;
	if(a.minutos!=b.minutos) return false;
	return true;
}

string practicaToString(Practica x)
{
	char sep = 2;
	string sIdPrac=to_string(x.idPrac);
	string sDescr=x.descr;
	return sIdPrac+sep+sDescr;
}

Practica practicaFromString(string s)
{
	char sep = 2;
	Practica x;
	string t0 = getTokenAt(s,sep,0);
	x.idPrac=stoi(t0);
	string t1 = getTokenAt(s,sep,1);

}
